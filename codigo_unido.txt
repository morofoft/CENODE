<script>
const DEPTS = ["Electrocardiograma","Glucometría","Rayos X","Mamografía","Densitometría","Sonografía","Doppler","Holter","MAPA","Ecocardiograma","Biopsia","OCT"];

let auditData = [];
let chartC = null, chartE = null;

function populateSelects() {
  ['f-dept','a-dept','r-dept'].forEach(id => {
    const el = document.getElementById(id);
    if (!el) return;
    DEPTS.forEach(d => {
      const opt = document.createElement('option');
      opt.value = d; opt.textContent = d;
      el.appendChild(opt);
    });
  });
}

function showTab(t, btn) {
  document.querySelectorAll('.tab').forEach(el => el.classList.remove('active'));
  document.querySelectorAll('.nav-btn').forEach(el => el.classList.remove('active'));
  document.getElementById('tab-' + t).classList.add('active');
  if (btn) btn.classList.add('active');
  if (t === 'dashboard') renderDashboard();
  if (t === 'auditoria') renderRecent();
}

function getFilteredData() {
  let data = [...auditData];
  const desde = document.getElementById('f-desde').value;
  const hasta = document.getElementById('f-hasta').value;
  const dept = document.getElementById('f-dept').value;
  if (desde) data = data.filter(r => r.fecha >= desde);
  if (hasta) data = data.filter(r => r.fecha <= hasta);
  if (dept) data = data.filter(r => r.departamento === dept);
  return data;
}

function resetFilter() {
  document.getElementById('f-desde').value = '';
  document.getElementById('f-hasta').value = '';
  document.getElementById('f-dept').value = '';
  renderDashboard();
}

function renderDashboard() {
  const data = getFilteredData();
  const total = data.reduce((s,r) => s + r.total, 0);
  const env = data.reduce((s,r) => s + r.enviadook, 0);
  const noenv = data.reduce((s,r) => s + r.noenv, 0);
  const monto = data.reduce((s,r) => s + r.monto, 0);
  const pct = total > 0 ? Math.round((env / total) * 100) : null;

  document.getElementById('kpi-total').textContent = total.toLocaleString();
  document.getElementById('kpi-enviado').textContent = env.toLocaleString();
  document.getElementById('kpi-noenviado').textContent = noenv.toLocaleString();
  document.getElementById('kpi-cumplimiento').textContent = pct !== null ? pct + '%' : '—';
  document.getElementById('kpi-cumplimiento').className = 'kpi-value ' + (pct === null ? '' : pct >= 80 ? 'green' : pct >= 60 ? 'amber' : 'red');
  document.getElementById('kpi-monto').textContent = '$' + monto.toLocaleString('en-US', {maximumFractionDigits: 0});
  document.getElementById('last-update').textContent = 'Actualizado: ' + new Date().toLocaleTimeString();

  renderDeptTable(data);
  renderCharts(data);
}

function renderDeptTable(data) {
  const deptMap = {};
  DEPTS.forEach(d => { deptMap[d] = {total:0,factok:0,realok:0,resultok:0,enviadook:0,noenv:0,monto:0}; });
  data.forEach(r => {
    const dm = deptMap[r.departamento];
    if (!dm) return;
    dm.total += r.total; dm.factok += r.factok; dm.realok += r.realok;
    dm.resultok += r.resultok; dm.enviadook += r.enviadook; dm.noenv += r.noenv; dm.monto += r.monto;
  });

  const tbody = document.getElementById('dept-table-body');
  tbody.innerHTML = '';
  let hasRows = false;
  DEPTS.forEach(d => {
    const dm = deptMap[d];
    if (dm.total === 0) return;
    hasRows = true;
    const pct = Math.round((dm.enviadook / dm.total) * 100);
    const cls = pct >= 80 ? 'pct-high' : pct >= 60 ? 'pct-mid' : 'pct-low';
    const fillColor = pct >= 80 ? '#3B6D11' : pct >= 60 ? '#BA7517' : '#D85A30';

function renderCharts(data) {
  const deptMap = {};
  DEPTS.forEach(d => { deptMap[d] = {total:0,enviadook:0}; });
  data.forEach(r => {
    if (deptMap[r.departamento]) {
      deptMap[r.departamento].total += r.total;
      deptMap[r.departamento].enviadook += r.enviadook;
    }
  });

  const deptLabels = DEPTS.filter(d => deptMap[d].total > 0);
  const deptPcts = deptLabels.map(d => {
    const dm = deptMap[d];
    return dm.total > 0 ? Math.round((dm.enviadook / dm.total) * 100) : 0;
  });
  const colors = deptPcts.map(p => p >= 80 ? '#3B6D11' : p >= 60 ? '#BA7517' : '#D85A30');

  if (chartC) chartC.destroy();
  chartC = new Chart(document.getElementById('chartCumplimiento'), {
    type: 'bar',
    data: {
      labels: deptLabels.length ? deptLabels.map(d => d.length > 9 ? d.substr(0,9)+'…' : d) : ['Sin datos'],
      datasets: [{
        label: 'Cumplimiento %',
        data: deptLabels.length ? deptPcts : [0],
        backgroundColor: deptLabels.length ? colors : ['#ccc'],
        borderRadius: 4
      }]
    },
    options: {
      responsive: true, maintainAspectRatio: false,
      plugins: { legend: { display: false } },
      scales: {
        y: { min:0, max:100, ticks:{ callback: v => v+'%', font:{size:10} }, grid:{color:'rgba(0,0,0,0.05)'} },
        x: { ticks:{ font:{size:9} }, grid:{display:false} }
      }
    }
  });

  const dateMap = {};
  data.forEach(r => { dateMap[r.fecha] = (dateMap[r.fecha] || 0) + r.enviadook; });
  const sortedDates = Object.keys(dateMap).sort().slice(-14);

  if (chartE) chartE.destroy();
  chartE = new Chart(document.getElementById('chartEvolucion'), {
    type: 'line',
    data: {
      labels: sortedDates.length ? sortedDates.map(d => d.substr(5)) : ['Sin datos'],
      datasets: [{
        label: 'Enviado OK',
        data: sortedDates.length ? sortedDates.map(d => dateMap[d]) : [0],
        borderColor: '#0F6E56',
        backgroundColor: 'rgba(15,110,86,0.08)',
        fill: true, tension: 0.4, pointRadius: 3
      }]
    },
    options: {
      responsive: true, maintainAspectRatio: false,
      plugins: { legend: { display: false } },
      scales: {
        y: { ticks:{ font:{size:10} }, grid:{color:'rgba(0,0,0,0.05)'} },
        x: { ticks:{ font:{size:9}, autoSkip: false, maxRotation: 45 }, grid:{display:false} }
      }
    }
  });
}

}

function saveAudit() {
  const dept = document.getElementById('a-dept').value;
  const fecha = document.getElementById('a-fecha').value;
  const total = parseInt(document.getElementById('a-total').value) || 0;
  const enviadook = parseInt(document.getElementById('a-enviadook').value) || 0;
  if (!dept || !fecha || total === 0) { showToast('Complete los campos obligatorios (*)'); return; }
  const record = {
    id: Date.now().toString(36),
    fecha, departamento: dept, total,
    factok: parseInt(document.getElementById('a-factok').value) || 0,
    realok: parseInt(document.getElementById('a-realok').value) || 0,
    resultok: parseInt(document.getElementById('a-resultok').value) || 0,
    enviadook,
    noenv: parseInt(document.getElementById('a-noenv').value) || 0,
    monto: parseFloat(document.getElementById('a-monto').value) || 0,
    cumplimiento: total > 0 ? parseFloat(((enviadook / total) * 100).toFixed(1)) : 0
  };
  auditData.unshift(record);
  clearForm();
  renderRecent();
  showToast('Auditoría guardada correctamente');
}

  auditData = auditData.filter(r => r.id !== id);

  if (document.getElementById('tab-dashboard').classList.contains('active')) renderDashboard();

  const tbody = document.getElementById('recent-table-body');

    const cls = pct >= 80 ? 'pct-high' : pct >= 60 ? 'pct-mid' : 'pct-low';

    tr.innerHTML = `

    tbody.appendChild(tr);
  });
}

function generateReport() {
  const tipo = document.getElementById('r-tipo').value;
  const dept = document.getElementById('r-dept').value;
  let data = dept ? auditData.filter(r => r.departamento === dept) : [...auditData];

  const output = document.getElementById('report-output');
  const title = document.getElementById('report-output-title');
  const content = document.getElementById('report-output-content');

  if (data.length === 0) { showToast('No hay datos para generar el reporte'); return; }

  output.style.display = 'block';
  title.textContent = tipo;

  const deptMap = {};
  DEPTS.forEach(d => { deptMap[d] = {total:0,enviadook:0,noenv:0,monto:0,count:0}; });
  data.forEach(r => {
    if (!deptMap[r.departamento]) return;
    deptMap[r.departamento].total += r.total;
    deptMap[r.departamento].enviadook += r.enviadook;
    deptMap[r.departamento].noenv += r.noenv;
    deptMap[r.departamento].monto += r.monto;
    deptMap[r.departamento].count++;
  });

  let html = '<div class="table-wrap"><table class="audit-table"><thead><tr><th>Departamento</th><th>Registros</th><th>Total</th><th>Enviado OK</th><th>No Enviado</th><th>Monto</th><th>Cumplimiento</th></tr></thead><tbody>';
  let hasRows = false;
  DEPTS.forEach(d => {
    const dm = deptMap[d];
    if (dm.total === 0) return;
    hasRows = true;
    const pct = Math.round((dm.enviadook / dm.total) * 100);
    const cls = pct >= 80 ? 'pct-high' : pct >= 60 ? 'pct-mid' : 'pct-low';
    html += `<tr><td>${d}</td><td>${dm.count}</td><td>${dm.total.toLocaleString()}</td><td>${dm.enviadook.toLocaleString()}</td><td>${dm.noenv.toLocaleString()}</td><td>$${dm.monto.toLocaleString('en-US',{maximumFractionDigits:0})}</td><td><span class="badge-pct ${cls}">${pct}%</span></td></tr>`;
  });
  html += '</tbody></table></div>';
  content.innerHTML = hasRows ? html : '<div class="empty-state"><div>No hay datos disponibles.</div></div>';

  const histEl = document.getElementById('report-history');
  const firstChild = histEl.querySelector('.empty-state');
  if (firstChild) firstChild.remove();

  const entry = document.createElement('div');
  entry.className = 'report-card';
  entry.innerHTML = `
    <div class="report-info">
      <div class="report-title">${tipo}${dept ? ' — ' + dept : ''}</div>
      <div class="report-date">Generado: ${new Date().toLocaleString()}</div>
    </div>
    <div><button class="btn-sec" onclick="this.closest('.report-card').remove()">Eliminar</button></div>`;
  histEl.prepend(entry);
  showToast('Reporte generado');
}

    data.forEach(r => {
      if (!map[r.departamento]) {
        map[r.departamento] = {
          total:0, env:0, monto:0
        };
      }
  
      map[r.departamento].total += r.total;
      map[r.departamento].env += r.enviadook;
      map[r.departamento].monto += r.monto;
    });
  
    // 🧾 TABLA
    let html = "<table border='1' cellpadding='6'><tr><th>Dept</th><th>Total</th><th>Enviado</th><th>%</th><th>Monto</th></tr>";
  
    for (let d in map) {
      const t = map[d].total;
      const e = map[d].env;
      const m = map[d].monto;
      const p = t > 0 ? Math.round((e/t)*100) : 0;
  
      html += `<tr>
        <td>${d}</td>
        <td>${t}</td>
        <td>${e}</td>
        <td>${p}%</td>
        <td>$${m.toLocaleString()}</td>
      </tr>`;
    }
  
    html += "</table>";
  
    document.getElementById("report-table").innerHTML = html;
  }

  function renderRecent() {
    const tbody = document.getElementById("recent-table-body");
    const empty = document.getElementById("recent-empty");
  
    if (!tbody) return;
  
    tbody.innerHTML = "";
  
    if (auditData.length === 0) {
      empty.style.display = "block";
      return;
    }
  
    empty.style.display = "none";
  
    auditData.slice().reverse().forEach(r => {
      const pct = Math.round(r.cumplimiento);
  
      const tr = document.createElement("tr");
  
      tr.innerHTML = `
        <td>${r.fecha}</td>
        <td>${r.departamento}</td>
        <td>${r.total}</td>
        <td>${r.enviadook}</td>
        <td>$${r.monto.toLocaleString()}</td>
        <td>${pct}%</td>
        <td>
          <button onclick="deleteRecord(${r.id})">✕</button>
        </td>
      `;
  
      tbody.appendChild(tr);
    });
  }

  function deleteRecord(id) {
    auditData = auditData.filter(r => r.id !== id);
    saveStorage();
    renderRecent();
  }

  let chartDept = null;
let chartEvolucion = null;


function renderCharts(data) {

    // =========================
    // 📊 AGRUPAR POR DEPARTAMENTO
    // =========================
    const map = {};
  
    data.forEach(r => {
      if (!map[r.departamento]) {
        map[r.departamento] = { total: 0, env: 0 };
      }
  
      map[r.departamento].total += r.total;
      map[r.departamento].env += r.enviadook;
    });
  
    const labels = Object.keys(map);
    const valores = labels.map(d => {
      const t = map[d].total;
      const e = map[d].env;
      return t > 0 ? Math.round((e / t) * 100) : 0;
    });
  
    // 🔥 destruir gráfico anterior
    if (chartDept) chartDept.destroy();
  
    chartDept = new Chart(document.getElementById("chartDept"), {
      type: "bar",
      data: {
        labels: labels,
        datasets: [{
          label: "% Cumplimiento",
          data: valores
        }]
      },
      options: {
        responsive: true,
        scales: {
          y: {
            min: 0,
            max: 100,
            ticks: {
              callback: v => v + "%"
            }
          }
        }
      }
    });
  
  
    // =========================
    // 📈 EVOLUCIÓN DIARIA
    // =========================
    const dateMap = {};
  
    data.forEach(r => {
      if (!dateMap[r.fecha]) dateMap[r.fecha] = 0;
      dateMap[r.fecha] += r.enviadook;
    });
  
    const fechas = Object.keys(dateMap).sort();
    const valoresFecha = fechas.map(f => dateMap[f]);
  
    if (chartEvolucion) chartEvolucion.destroy();
  
    chartEvolucion = new Chart(document.getElementById("chartEvolucion"), {
      type: "line",
      data: {
        labels: fechas,
        datasets: [{
          label: "Enviado OK",
          data: valoresFecha,
          fill: true
        }]
      },
      options: {
        responsive: true
      }
    });
  }
